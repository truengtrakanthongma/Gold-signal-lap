ใส่ Gold Signal Lab ใน TradingView

ได้ราคา XAUUSD จริงจากโบรกเกอร์คุณ ไม่มีโควตา ทดสอบย้อนหลังในตัว และแจ้งเตือนเข้ามือถือ

ข้อจำกัดที่ต้องรู้ก่อน: แอป TradingView บนมือถือไม่มี Pine Editor จึงวางโค้ดในแอปไม่ได้ ต้องวางผ่านเบราว์เซอร์ในโหมดคอมพิวเตอร์ ครั้งเดียว หลังจากนั้นใช้ในแอปได้ตามปกติ รวมถึงรับแจ้งเตือน
  1. กดปุ่ม คัดลอกโค้ดทั้งหมด ข้างบน
  2. เปิด Chrome (Android) หรือ Safari (iPhone) — ไม่ใช่แอป TradingView
  3. เปิดโหมดคอมพิวเตอร์:
    • Chrome → จุดสามจุดมุมขวาบน → ติ๊ก เว็บไซต์เดสก์ท็อป
    • Safari → ไอคอน ᴀA ซ้ายช่อง URL → ขอเว็บไซต์เดสก์ท็อป
  4. ไปที่ tradingview.com/chart แล้วเข้าสู่ระบบ (สมัครฟรีได้)
  5. ค้นหาสัญลักษณ์ XAUUSD แล้วเลือกกรอบเวลา 15 นาที
  6. ล่างจอมีแถบ Pine Editor — แตะเพื่อกางขึ้นมา
    หมุนมือถือเป็นแนวนอนตรงนี้จะง่ายขึ้นมาก
  7. ลบโค้ดตัวอย่างที่มีอยู่ให้หมด แล้ววางโค้ดที่คัดลอกมา
  8. กด Save → ตั้งชื่อ Gold Signal Lab → กด Add to chart
  9. เสร็จแล้ว! ปิดโหมดเดสก์ท็อปได้ กลับไปใช้แอป TradingView ตามปกติ — สคริปต์จะอยู่ใน Indicators → My scripts เพิ่มลงกราฟได้เลย

  7. เปิดแท็บ Strategy Tester ล่างจอ เพื่อดูผลทดสอบย้อนหลังกับทองจริง

ตั้งแจ้งเตือนเข้ามือถือ

  1. บนกราฟที่ใส่สคริปต์แล้ว กดไอคอน นาฬิกาปลุก (Alert)
  2. ช่อง Condition เลือก Gold Signal Lab
  3. เลือกเงื่อนไข Any alert() function call
  4. ในแท็บ Notifications ติ๊ก Push notification
  5. กด Create — จากนี้สัญญาณจะเด้งเข้ามือถือเอง ไม่ต้องเปิดจอทิ้งไว้

อ่านผลทดสอบย้อนหลังยังไง

ในแท็บ Strategy Tester ดูสามตัวนี้พอ:
• Profit Factor — เกิน 1.0 คือกำไร ต่ำกว่าคือขาดทุน
• Total Closed Trades — ต่ำกว่า 30 ไม้ ยังสรุปอะไรไม่ได้ ให้ถือว่าเป็นแค่ตัวอย่าง
• Max Drawdown — เคยติดลบลึกสุดเท่าไร ถ้าคุณทนไม่ไหว ก็ใช้ระบบนี้ไม่ได้ ต่อให้สุดท้ายมันกำไร
ค่าเริ่มต้นเป็นค่ากลาง ๆ ไม่ได้จูนมาเพื่อทองคำโดยเฉพาะ ถ้าผลออกมาไม่ดี อย่าเพิ่งสรุปว่ากฎใช้ไม่ได้ — ลองปรับ คะแนนขั้นต่ำ และ เป้าทำกำไร ในหน้าตั้งค่าของสคริปต์ดูก่อน และอย่าเชื่อค่าที่จูนจนสวยที่สุด เพราะนั่นคือการฟิตกับอดีต

โค้ดทั้งหมด

ถ้าปุ่มคัดลอกใช้ไม่ได้ กดค้างในกล่องนี้แล้วเลือกทั้งหมดเองได้

// © Gold Signal Lab — พอร์ตมาจากเว็บแอป https://github.com/truengtrakanthongma/Gold-signal-lap
//
// ทำไมต้องมีเวอร์ชันนี้: เว็บแอปต้องหาข้อมูลราคาเอง ซึ่งฟรีแล้วมีข้อจำกัดเสมอ
// รันใน TradingView แทน จะได้ข้อมูลจากโบรกเกอร์ของคุณเอง (XAUUSD จริง) ไม่มีโควตา
// พร้อมเครื่องทดสอบย้อนหลังในตัว และแจ้งเตือนเข้ามือถือผ่านแอป TradingView
//
// วิธีใช้: TradingView → Pine Editor → วางโค้ดนี้ → Add to chart
//
// ⚠ เครื่องมือเพื่อการศึกษา ไม่ใช่คำแนะนำการลงทุน
//@version=6
strategy("Gold Signal Lab", overlay = true,
     default_qty_type = strategy.fixed, default_qty_value = 1,
     initial_capital = 10000, currency = currency.USD,
     commission_type = strategy.commission.percent, commission_value = 0.01,
     calc_on_every_tick = false, process_orders_on_close = true)

// ───────────────────────── ตั้งค่า ─────────────────────────
gTune   = "เกณฑ์สัญญาณ"
thresh  = input.float(35.0, "คะแนนขั้นต่ำที่ถือว่าเป็นสัญญาณ", minval = 5, maxval = 95, step = 1, group = gTune)
adxMin  = input.float(22.0, "ADX เกินนี้ = ตลาดมีแนวโน้ม (ต่ำกว่า = ออกข้าง)", minval = 5, maxval = 50, group = gTune)
slMult  = input.float(1.5,  "จุดตัดขาดทุน = กี่เท่าของ ATR", minval = 0.3, maxval = 5, step = 0.1, group = gTune)
maxSl   = input.float(3.0,  "เพดานความกว้างจุดตัดขาดทุน (เท่าของ ATR)", minval = 1, maxval = 8, step = 0.5, group = gTune)
targetR = input.float(2.0,  "เป้าทำกำไร = กี่เท่าของความเสี่ยง", minval = 0.5, maxval = 6, step = 0.25, group = gTune)
riskPct = input.float(1.0,  "ความเสี่ยงต่อไม้ (% ของทุน)", minval = 0.1, maxval = 10, step = 0.1, group = gTune)

gFilt   = "ตัวกรอง"
useHtf  = input.bool(true,  "ต้องให้กรอบเวลาใหญ่เห็นตรงกัน", group = gFilt)
htfTf   = input.timeframe("240", "กรอบเวลาใหญ่", group = gFilt)
useVol  = input.bool(true,  "กรองช่วงผันผวนผิดปกติ", group = gFilt)
minAtrP = input.float(0.02, "ผันผวนต่ำสุดที่ยอมเทรด (% ของราคา)", minval = 0, step = 0.01, group = gFilt)
maxAtrP = input.float(1.5,  "ผันผวนสูงสุดที่ยอมเทรด (% ของราคา)", minval = 0.1, step = 0.1, group = gFilt)
useSess = input.bool(false, "เทรดเฉพาะช่วงเวลาที่กำหนด", group = gFilt)
sessStr = input.session("1400-2400", "ช่วงเวลา (เวลาไทย)", group = gFilt)

gW      = "น้ำหนักปัจจัย (รวม 120)"
wEma    = input.float(16, "เส้นค่าเฉลี่ยเรียงตัว",   minval = 0, maxval = 40, group = gW)
wAdx    = input.float(12, "ความแรงแนวโน้ม",        minval = 0, maxval = 40, group = gW)
wMacd   = input.float(12, "แรงส่งของราคา",          minval = 0, maxval = 40, group = gW)
wRsi    = input.float(10, "แรงซื้อ-แรงขาย",         minval = 0, maxval = 40, group = gW)
wStruct = input.float(12, "โครงสร้างราคา",          minval = 0, maxval = 40, group = gW)
wPat    = input.float(10, "รูปแบบแท่งเทียน",        minval = 0, maxval = 40, group = gW)
wVolume = input.float(8,  "ปริมาณซื้อขายยืนยัน",     minval = 0, maxval = 40, group = gW)
wBands  = input.float(8,  "กรอบความผันผวน",         minval = 0, maxval = 40, group = gW)
wLevels = input.float(10, "แนวรับ-แนวต้าน",         minval = 0, maxval = 40, group = gW)
wDiv    = input.float(10, "สัญญาณแรงหมด",           minval = 0, maxval = 40, group = gW)
wVwap   = input.float(6,  "เทียบต้นทุนเฉลี่ยของวัน", minval = 0, maxval = 40, group = gW)
wStoch  = input.float(6,  "จุดตัดระยะสั้น",          minval = 0, maxval = 40, group = gW)

// ───────────────────────── ตัวชี้วัด ─────────────────────────
ema20  = ta.ema(close, 20)
ema50  = ta.ema(close, 50)
ema200 = ta.ema(close, 200)
rsi    = ta.rsi(close, 14)
atr    = ta.atr(14)
atrPct = atr / close * 100
[_macdLine, _macdSig, macdHist] = ta.macd(close, 12, 26, 9)
[bbMid, bbUp, bbLo] = ta.bb(close, 20, 2.0)
[diPlus, diMinus, adx] = ta.dmi(14, 14)
stochK = ta.sma(ta.stoch(close, high, low, 14), 3)
stochD = ta.sma(stochK, 3)
vwap   = ta.vwap(hlc3)
volSma = ta.sma(volume, 20)
bbWidth = (bbUp - bbLo) / bbMid * 100
squeeze = bbWidth <= ta.lowest(bbWidth, 20) * 1.08

isTrend = adx >= adxMin

// โครงสร้างราคา — pivot ยืนยันช้า 3 แท่ง เหมือนเวอร์ชันเว็บ (ไม่มองอนาคต)
ph = ta.pivothigh(high, 3, 3)
pl = ta.pivotlow(low, 3, 3)
var float lastPH = na, var float prevPH = na
var float lastPL = na, var float prevPL = na
if not na(ph)
    prevPH := lastPH
    lastPH := ph
if not na(pl)
    prevPL := lastPL
    lastPL := pl
hh = not na(lastPH) and not na(prevPH) and lastPH > prevPH
hl = not na(lastPL) and not na(prevPL) and lastPL > prevPL
lh = not na(lastPH) and not na(prevPH) and lastPH < prevPH
ll = not na(lastPL) and not na(prevPL) and lastPL < prevPL
structSide = hh and hl ? 1 : lh and ll ? -1 : 0

// ───────────────────────── ให้คะแนนแต่ละปัจจัย ─────────────────────────
// แต่ละตัวคืนค่า -1..+1 แล้วคูณน้ำหนัก เหมือนเวอร์ชันเว็บ
sEma = close > ema20 and ema20 > ema50 and ema50 > ema200 ? 1.0 :
       close < ema20 and ema20 < ema50 and ema50 < ema200 ? -1.0 :
       close > ema20 and ema20 > ema50 ? 0.7 :
       close < ema20 and ema20 < ema50 ? -0.7 :
       close > ema50 ? 0.35 : close < ema50 ? -0.35 : 0.0

adxStrength = math.min(1.0, math.max(0.2, (adx - 18) / 27))
sAdx = isTrend ? (diPlus > diMinus ? adxStrength : -adxStrength) : 0.0

macdRising = macdHist > macdHist[1]
sMacd = macdHist > 0 and macdRising ? math.max(0.45, math.min(1.0, math.abs(macdHist) / (atr * 0.6))) :
        macdHist < 0 and not macdRising ? -math.max(0.45, math.min(1.0, math.abs(macdHist) / (atr * 0.6))) :
        macdHist > 0 ? 0.2 : macdHist < 0 ? -0.2 : 0.0

// RSI ตีความคนละแบบตามสภาพตลาด — จุดที่ระบบง่าย ๆ มักพลาด
sRsi = isTrend ?
       (rsi > 55 and rsi < 78 ? math.min(1.0, (rsi - 50) / 25) :
        rsi < 45 and rsi > 22 ? -math.min(1.0, (50 - rsi) / 25) :
        rsi >= 78 ? -0.3 : rsi <= 22 ? 0.3 : 0.0) :
       (rsi < 32 ? math.min(1.0, math.max(0.3, (32 - rsi) / 20)) :
        rsi > 68 ? -math.min(1.0, math.max(0.3, (rsi - 68) / 20)) : 0.0)

sStruct = structSide * 0.85

// รูปแบบแท่งเทียน
body = math.abs(close - open)
rng  = math.max(high - low, syminfo.mintick)
upW  = high - math.max(close, open)
dnW  = math.min(close, open) - low
bullEng = close >= open and close[1] < open[1] and close > open[1] and open <= close[1] and body > math.abs(close[1] - open[1])
bearEng = close < open and close[1] >= open[1] and close < open[1] and open >= close[1] and body > math.abs(close[1] - open[1])
hammer  = dnW > body * 2 and dnW / rng > 0.55 and upW < body
star    = upW > body * 2 and upW / rng > 0.55 and dnW < body
bigBull = body > atr * 1.3 and close >= open
bigBear = body > atr * 1.3 and close < open
sPat = bullEng or hammer ? 0.9 : bearEng or star ? -0.9 : bigBull ? 0.6 : bigBear ? -0.6 : 0.0

volRatio = volSma > 0 ? volume / volSma : 1.0
volDir   = close >= open ? 1 : -1
sVol = volRatio > 1.25 ? volDir * math.min(1.0, math.max(0.3, (volRatio - 1) / 1.5)) : 0.0

pctB = bbUp > bbLo ? (close - bbLo) / (bbUp - bbLo) : 0.5
sBands = squeeze and close > bbUp ? 0.95 :
         squeeze and close < bbLo ? -0.95 :
         not isTrend and pctB <= 0.05 ? 0.7 :
         not isTrend and pctB >= 0.95 ? -0.7 :
         isTrend and pctB > 0.75 ? 0.35 :
         isTrend and pctB < 0.25 ? -0.35 : 0.0

// แนวรับ-ต้านจาก pivot ที่ยืนยันแล้ว
nearSup = not na(lastPL) and (close - lastPL) / atr < 0.6 and structSide >= 0
nearRes = not na(lastPH) and (lastPH - close) / atr < 0.6 and structSide <= 0
brokeRes = not na(lastPH) and close > lastPH and close[1] <= lastPH
brokeSup = not na(lastPL) and close < lastPL and close[1] >= lastPL
sLevels = brokeRes ? 0.85 : brokeSup ? -0.85 : nearSup ? 0.6 : nearRes ? -0.6 : 0.0

// สัญญาณแรงหมด: ราคาทำจุดใหม่แต่ RSI ไม่ตาม
var float rsiAtLastPL = na, var float rsiAtPrevPL = na
var float rsiAtLastPH = na, var float rsiAtPrevPH = na
if not na(pl)
    rsiAtPrevPL := rsiAtLastPL
    rsiAtLastPL := rsi[3]
if not na(ph)
    rsiAtPrevPH := rsiAtLastPH
    rsiAtLastPH := rsi[3]
bullDiv = not na(prevPL) and lastPL < prevPL and not na(rsiAtPrevPL) and rsiAtLastPL > rsiAtPrevPL + 2
bearDiv = not na(prevPH) and lastPH > prevPH and not na(rsiAtPrevPH) and rsiAtLastPH < rsiAtPrevPH - 2
sDiv = bullDiv ? 0.85 : bearDiv ? -0.85 : 0.0

vwapDev = (close - vwap) / atr
sVwap = math.abs(vwapDev) > 0.15 ? math.sign(vwapDev) * math.min(1.0, math.max(0.2, math.abs(vwapDev) / 2)) : 0.0

sStoch = ta.crossover(stochK, stochD) and stochK < 45 ? 0.8 :
         ta.crossunder(stochK, stochD) and stochK > 55 ? -0.8 : 0.0

// ───────────────────────── รวมคะแนน ─────────────────────────
totalW = wEma + wAdx + wMacd + wRsi + wStruct + wPat + wVolume + wBands + wLevels + wDiv + wVwap + wStoch
raw = sEma * wEma + sAdx * wAdx + sMacd * wMacd + sRsi * wRsi + sStruct * wStruct +
      sPat * wPat + sVol * wVolume + sBands * wBands + sLevels * wLevels +
      sDiv * wDiv + sVwap * wVwap + sStoch * wStoch
score = totalW > 0 ? raw / totalW * 100 : 0.0

// กรอบเวลาใหญ่ — ใช้แค่ทิศทางของเส้นค่าเฉลี่ย เพื่อไม่ให้เรียกซ้ำหนักเกินไป
htfEma20 = request.security(syminfo.tickerid, htfTf, ta.ema(close, 20), lookahead = barmerge.lookahead_off)
htfEma50 = request.security(syminfo.tickerid, htfTf, ta.ema(close, 50), lookahead = barmerge.lookahead_off)
htfBull = htfEma20 > htfEma50
htfBear = htfEma20 < htfEma50
htfOk = not useHtf or (score > 0 and htfBull) or (score < 0 and htfBear)

volOk  = not useVol or (atrPct >= minAtrP and atrPct <= maxAtrP)
sessOk = not useSess or not na(time(timeframe.period, sessStr, "Asia/Bangkok"))
canTrade = htfOk and volOk and sessOk and barstate.isconfirmed

longSig  = score >=  thresh and canTrade
shortSig = score <= -thresh and canTrade

// ───────────────────────── แผนเข้า-ออก ─────────────────────────
// จุดตัดขาดทุน: ใช้ค่าที่กว้างกว่าระหว่าง ATR กับเลย swing ล่าสุด แต่ไม่เกินเพดาน
slDistLong  = math.min(atr * maxSl, math.max(atr * slMult, na(lastPL) ? 0 : close - lastPL + atr * 0.15))
slDistShort = math.min(atr * maxSl, math.max(atr * slMult, na(lastPH) ? 0 : lastPH - close + atr * 0.15))

qtyFor(slDist) =>
    riskMoney = strategy.equity * riskPct / 100
    slDist > 0 ? riskMoney / slDist : 0.0

qLong  = qtyFor(slDistLong)
qShort = qtyFor(slDistShort)

if longSig and strategy.position_size == 0 and qLong > 0 and slDistLong > 0
    strategy.entry("ซื้อ", strategy.long, qty = qLong)
    strategy.exit("ออก-ซื้อ", "ซื้อ", stop = close - slDistLong, limit = close + slDistLong * targetR)

if shortSig and strategy.position_size == 0 and qShort > 0 and slDistShort > 0
    strategy.entry("ขาย", strategy.short, qty = qShort)
    strategy.exit("ออก-ขาย", "ขาย", stop = close + slDistShort, limit = close - slDistShort * targetR)

// ───────────────────────── แสดงผล ─────────────────────────
plot(ema20,  "EMA20",  color.new(#facc15, 0), 1)
plot(ema50,  "EMA50",  color.new(#38bdf8, 0), 1)
plot(ema200, "EMA200", color.new(#f472b6, 0), 2)
plot(vwap,   "VWAP",   color.new(#a78bfa, 40), 1)

plotshape(longSig  and strategy.position_size == 0, "สัญญาณซื้อ", shape.triangleup,
     location.belowbar, color.new(color.green, 0), size = size.small)
plotshape(shortSig and strategy.position_size == 0, "สัญญาณขาย", shape.triangledown,
     location.abovebar, color.new(color.red, 0), size = size.small)

bgcolor(longSig ? color.new(color.green, 88) : shortSig ? color.new(color.red, 88) : na)

// ตารางสรุปมุมขวาบน — อ่านได้ทันทีว่าตอนนี้ระบบคิดยังไง
var table info = table.new(position.top_right, 2, 6, border_width = 1)
if barstate.islast
    table.cell(info, 0, 0, "คะแนน",       text_color = color.gray, text_size = size.small)
    table.cell(info, 1, 0, str.tostring(score, "#.0"),
         text_color = score > 0 ? color.green : score < 0 ? color.red : color.gray, text_size = size.small)
    table.cell(info, 0, 1, "สภาพตลาด",    text_color = color.gray, text_size = size.small)
    table.cell(info, 1, 1, isTrend ? "มีแนวโน้ม" : "ออกข้าง", text_color = color.white, text_size = size.small)
    table.cell(info, 0, 2, "ADX",          text_color = color.gray, text_size = size.small)
    table.cell(info, 1, 2, str.tostring(adx, "#.1"), text_color = color.white, text_size = size.small)
    table.cell(info, 0, 3, "ATR (%)",      text_color = color.gray, text_size = size.small)
